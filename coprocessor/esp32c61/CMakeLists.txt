cmake_minimum_required(VERSION 3.16)
set(PROJECT_VER "0.1.0-dev")
include($ENV{IDF_PATH}/tools/cmake/version.cmake)
if(NOT IDF_VERSION_MAJOR EQUAL 5 OR NOT IDF_VERSION_MINOR EQUAL 5 OR
   NOT IDF_VERSION_PATCH EQUAL 5)
    message(FATAL_ERROR "Paired C61 firmware requires ESP-IDF 5.5.5")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(xiaotai_wt99p461_c61)
if(NOT CONFIG_IDF_TARGET_ESP32C61 OR NOT CONFIG_ESP_HOSTED_CP OR
   NOT CONFIG_EH_TRANSPORT_CP_SPI OR
   NOT CONFIG_EH_TRANSPORT_CP_SPI_HSPI_GPIO_MOSI EQUAL 27 OR
   NOT CONFIG_EH_TRANSPORT_CP_SPI_HSPI_GPIO_MISO EQUAL 28 OR
   NOT CONFIG_EH_TRANSPORT_CP_SPI_HSPI_GPIO_CLK EQUAL 26 OR
   NOT CONFIG_EH_TRANSPORT_CP_SPI_HSPI_GPIO_CS EQUAL 25 OR
   NOT CONFIG_EH_TRANSPORT_CP_SPI_GPIO_HANDSHAKE EQUAL 22 OR
   NOT CONFIG_EH_TRANSPORT_CP_SPI_GPIO_DATA_READY EQUAL 23 OR
   NOT CONFIG_EH_TRANSPORT_CP_SPI_MODE_3)
    message(FATAL_ERROR "C61 paired SPI transport/pins do not match WT01P461 module")
endif()
